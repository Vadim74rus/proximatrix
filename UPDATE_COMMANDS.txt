# Команды для обновления из GitHub и перезапуска
# Выполняйте команды по порядку на сервере через PuTTY

# ============================================
# ШАГ 1: Остановка контейнеров
# ============================================

cd /opt/proximatrix
docker compose down

# ============================================
# ШАГ 2: Скачивание изменений из GitHub
# ============================================

git pull

# ============================================
# ШАГ 3: Пересборка контейнеров (опционально)
# ============================================

docker compose build --no-cache

# ============================================
# ШАГ 4: Запуск контейнеров
# ============================================

docker compose up -d

# ============================================
# ШАГ 5: Проверка статуса
# ============================================

docker compose ps

# ============================================
# ШАГ 6: Просмотр логов
# ============================================

docker compose logs -f proximatrix

# (Нажмите Ctrl+C для выхода из логов)
