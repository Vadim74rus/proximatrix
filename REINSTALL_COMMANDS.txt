# Команды для полной переустановки Proximatrix
# Выполняйте команды по порядку на сервере через PuTTY

# ============================================
# ШАГ 1: Остановка старых контейнеров
# ============================================

cd /opt/proximatrix
docker compose down
docker rm -f proximatrix-proxy proximatrix-mongo 2>/dev/null || true

# ============================================
# ШАГ 2: Удаление старой директории
# ============================================

cd /opt
rm -rf proximatrix

# ============================================
# ШАГ 3: Клонирование из GitHub
# ============================================

git clone https://github.com/Vadim74rus/proximatrix.git proximatrix

# ============================================
# ШАГ 4: Переход в директорию проекта
# ============================================

cd /opt/proximatrix
ls -la

# ============================================
# ШАГ 5: Настройка файрвола (UFW)
# ============================================

ufw allow 8444/tcp comment 'MTProto для Telegram'
ufw allow 1080/tcp comment 'SOCKS5 для Telegram'
ufw allow 8080/tcp comment 'HTTP для WhatsApp'
ufw allow 8443/tcp comment 'HTTPS для WhatsApp'
ufw allow 9090/tcp comment 'API управления пользователями'
ufw --force enable
ufw status

# ============================================
# ШАГ 6: Сборка Docker контейнеров
# ============================================

cd /opt/proximatrix
docker compose build --no-cache

# ============================================
# ШАГ 7: Запуск контейнеров
# ============================================

docker compose up -d

# ============================================
# ШАГ 8: Проверка статуса
# ============================================

docker compose ps

# ============================================
# ШАГ 9: Просмотр логов
# ============================================

docker compose logs -f proximatrix

# (Нажмите Ctrl+C для выхода из логов)

# ============================================
# ШАГ 10: Проверка API
# ============================================

curl -H "X-API-Key: proximatrix-api-key-change-in-production" http://localhost:9090/api/users

# ============================================
# ГОТОВО!
# ============================================
